% 生成画像は、要点タイトル・図中ラベル・結論までを一枚のスライドとして完結させます。
% TeX側で説明帯を追加せず、画像の構造と文字の対応をそのまま保ちます。
\newcommand{\chapterimage}[3]{%
  \begin{figure}[htbp]
    \centering
    \IfFileExists{assets/#1}{%
      \includegraphics[width=0.96\linewidth]{assets/#1}%
    }{%
      \PackageError{cs-doc}{Required image `assets/#1' is missing}{Add the generated chapter image.}%
    }
    \caption{#2}
    \label{#3}
  \end{figure}
}

\newcommand{\sectionimage}[3]{%
  \begin{figure}[htbp]
    \centering
    \IfFileExists{assets/#1}{%
      \includegraphics[width=0.90\linewidth]{assets/#1}%
    }{%
      \PackageError{cs-doc}{Required image `assets/#1' is missing}{Add the generated section image.}%
    }
    \caption{#2}
    \label{#3}
  \end{figure}
}

\begin{frame}[t]{プログラムは、OSに読み込まれてプロセスとして動きます}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{プログラム}
        命令や初期データを含む静的な記述です。実行可能ファイルやスクリプトは、それだけではCPU時間を持ちません。
      \end{block}
      \begin{block}{プロセス}
        実行中のプログラムに、アドレス空間、レジスタ状態、開いているファイルなどの実行文脈を加えたものです。
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{スレッド}
        プロセス内の実行の流れです。同じプロセスのスレッドはアドレス空間などを共有し、各スレッドはレジスタ状態とスタックを持ちます。
      \end{block}
      \begin{block}{並行性の注意}
        共有データへの読み書きが競合すると、実行順によって結果が変わります。同期で許される順序を定めます。
      \end{block}
  \end{columns}
  \takeaway{同じプログラムを2回起動すると、通常は別のプロセスと別の仮想アドレス空間が作られます。}
\end{frame}

\begin{frame}[t]{ポインタと参照は、値の所在と寿命を意識させます}
  \begin{columns}[T,onlytextwidth]
    \column{0.31\textwidth}
      \begin{block}{ポインタ／参照}
        別の値へ到達するための情報です。複数の変数が同じ対象を共有すると、一方の更新を他方から観測できます。
      \end{block}
    \column{0.31\textwidth}
      \begin{block}{寿命の誤り}
        解放済み領域を指す参照や、必要な値を早く解放する処理は不正アクセスにつながります。
      \end{block}
    \column{0.31\textwidth}
      \begin{block}{GC}
        ガベージコレクションは到達可能性などを基準に、不要になった領域を自動で再利用します。資源解放の時刻までは保証しません。
      \end{block}
  \end{columns}
  \vspace{4mm}
  \begin{block}{Goでの境界}
    Goはポインタを持ちますが、通常のポインタ演算は認めません。値をスタックかヒープのどちらへ置くかは、エスケープ解析などを用いる処理系の判断です。
  \end{block}
  \source{The Go Programming Language Specification; Go FAQ, “stack or heap”}
\end{frame}

\begin{frame}[t]{型は、値と許される操作の集合を定めます}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{型が表すもの}
        値の集合、利用できる演算、式の組み合わせ方を定めます。表現方法や実行時検査の有無は、言語仕様と実装を分けて確認します。
      \end{block}
      \begin{block}{型安全性の範囲}
        型が付けば業務上正しいとは限りません。例えば、同じ整数型でも円と個数の取り違えは防げません。
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{静的型付け}
        実行前の解析で式の型整合性を主に検査します。検査を通った後も、配列境界や0除算など別種の実行時エラーは残ります。
      \end{block}
      \begin{block}{動的型付け}
        実行時の値が型を持ち、操作時に適用可能性を検査します。型が存在しないという意味ではありません。
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}[t]{Goの基本型は、用途と表現範囲を明示します}
  \begin{columns}[T,onlytextwidth]

        \begin{tabularx}{\linewidth}{@{}lY@{}}
          \toprule

          \midrule
          \code{bool} & 真偽値\\

      \end{block}
      \begin{block}{ゼロ値}
        明示的に初期化していない変数にも、型ごとのゼロ値があります。数値は0、真偽値は\code{false}、文字列は空文字列です。
      \end{block}
  \end{columns}

\begin{frame}[t]{Goは値を渡しますが、値の内部で記憶領域を共有できます}

\begin{frame}[t]{計算量は、入力サイズが増えたときの増え方を表します}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
      \centering
      \begin{tikzpicture}[x=5mm,y=4mm,every node/.append style={font=\scriptsize}]
        \draw[->,draw=PaleGray] (0,0)--(8.5,0) node[right] {$n$};
        \draw[->,draw=PaleGray] (0,0)--(0,7) node[above] {操作回数};
        \draw[draw=DeepNavy,line width=.7pt] plot[smooth] coordinates {(0,1)(2,1)(4,1)(6,1)(8,1)};
        \node[right] at (8,1) {$O(1)$};
        \draw[draw=MutedBlue,line width=.7pt] plot[smooth] coordinates {(0,0)(2,1.4)(4,2.8)(6,4.2)(8,5.6)};
        \node[right] at (8,5.6) {$O(n)$};
        \draw[draw=PaleGray,line width=.7pt] plot[smooth] coordinates {(0,0)(2,.35)(4,1.4)(6,3.15)(8,5.6)};
        \node[above left] at (7.7,5.3) {$O(n^2)$};
      \end{tikzpicture}
    \column{0.44\textwidth}
      \begin{block}{入力サイズ $n$}
        要素数、頂点数、桁数など、処理対象の規模を表す量です。何を$n$とするかを先に定義します。
      \end{block}
      \begin{block}{O記法}
        十分大きな$n$で、増え方の上限を定数倍まで無視して表します。実測時間そのものではありません。
      \end{block}
  \end{columns}
  \vspace{3mm}
  \takeaway{同じ$O(n)$でも定数、メモリアクセス、入力分布、実装によって実時間は異なります。}
\end{frame}

\begin{frame}[t]{時間・空間と、最良・平均・最悪を分けて評価します}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{時間計算量}
        比較、代入、メモリアクセスなど、計算モデルで定めた基本操作の回数を数えます。
      \end{block}
      \begin{block}{空間計算量}
        入力以外に必要な補助記憶や、再帰スタックを含む使用量の増え方を評価します。
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{線形探索の例}
        \begin{tabularx}{\linewidth}{@{}lY@{}}
          \toprule
          条件 & 比較回数\\
          \midrule
          最良 & 先頭で一致：$1$\\
          平均 & 分布の仮定に依存\\
          最悪 & 末尾または不在：$n$\\
          \bottomrule
        \end{tabularx}
      \end{block}
      \begin{block}{平均には仮定が必要です}
        どの入力がどの確率で現れるかを示さない「平均」は再現できません。
      \end{block}
  \end{columns}
  \source{MIT 6.006, “Asymptotic Complexity”}
\end{frame}

\begin{frame}[t]{append後は、返されたスライスを基準に考えます}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{容量が足りる場合}
        元配列の空き領域へ要素を書き、同じ元配列を参照する可能性があります。別名のスライスにも影響し得ます。
      \end{block}
      \begin{block}{容量が足りない場合}
        より大きな元配列を確保して要素をコピーし、そこを指す新しいスライスを返します。元配列との共有は切れます。
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{必ず返り値を受け取ります}
        \code{s = append(s, x)}のように書きます。再確保の有無と容量の増やし方は、コードから固定値として仮定しません。
      \end{block}
      \begin{block}{償却$O(1)$}
        容量を幾何級数的に増やす動的配列では、複数回の末尾追加へ再確保コストを配ると1回当たり$O(1)$になります。これは一回ごとの最悪時間ではありません。
      \end{block}
  \end{columns}
  \source{Go specification, built-in \code{append}; MIT 6.006, “Dynamic Arrays”}
\end{frame}

\begin{frame}[t]{マップは、ハッシュ値からキーの格納候補を探します}

        \node[box,right=of hash] (idx) {候補 2};
        \draw[flow] (key)--(hash); \draw[flow] (hash)--(idx);
        \node[box,below=8mm of hash,minimum width=46mm] (slots) {0：\quad 1：\quad 2：cat／値\quad 3：dog／値};
        \draw[flow] (idx)--(slots);
      \end{tikzpicture}
      \vspace{2mm}
      \begin{block}{衝突}
        異なるキーが同じ候補へ対応することがあります。実装はキー比較と追加の探索で区別します。
      \end{block}
    \column{0.42\textwidth}
      \begin{block}{負荷率とリサイズ}
        衝突や探索が増え過ぎないよう、要素数に応じて格納領域を拡張し、配置を組み直す実装があります。
      \end{block}
      \begin{block}{計算量と保証範囲}
        良い分散と適切な負荷率を仮定すれば検索・追加は平均$O(1)$、偏れば最悪$O(n)$です。
        Go仕様は具体的な表構造や反復順を保証しません。
      \end{block}
  \end{columns}
  \source{CMU 15-121, “Hash Tables”; The Go Programming Language Specification}
\end{frame}

\begin{frame}[t]{データ構造は、必要な操作の組み合わせで選びます}
  \footnotesize
  \begin{tabularx}{\textwidth}{@{}lYYYY@{}}
    \toprule
    構造 & 添字／キー検索 & 末尾追加 & 途中挿入 & 主な注意点\\
    \midrule
    固定長配列 & $O(1)$ & 対象外 & $O(n)$ & 長さが型に含まれる\\
    動的配列／スライス & $O(1)$ & 償却$O(1)$ & $O(n)$ & 再確保と元配列共有\\
    ハッシュマップ & 平均$O(1)$ & 平均$O(1)$ & 対象外 & 衝突・順序なし\\
    連結リスト & $O(n)$ & 位置次第 & 位置取得後$O(1)$ & ポインタ追跡\\

      \end{block}
    \column{0.48\textwidth}
      \begin{block}{記号の前提}

      \end{block}

\begin{frame}[t]{Pythonでは、名前が型を持つオブジェクトへ束縛されます}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \centering
      \begin{tikzpicture}[node distance=6mm,every node/.append style={font=\scriptsize}]
        \node[box] (x) {名前 x};
        \node[box,below=of x] (y) {名前 y};
        \node[accentbox,right=13mm of $(x)!0.5!(y)$] (obj) {オブジェクト\\型：list\\値：[1, 2]};
        \draw[flow] (x)--(obj); \draw[flow] (y)--(obj);
      \end{tikzpicture}
      \vspace{4mm}
      \begin{block}{共有と可変性}
        \code{y = x}は同じオブジェクトへの束縛を作ります。リストを\code{y.append(3)}で変更すると、\code{x}からも変更後の値が見えます。
      \end{block}
    \column{0.46\textwidth}
      \begin{block}{オブジェクトモデル}
        Pythonのすべてのデータはオブジェクトで、各オブジェクトは同一性、型、値を持ちます。オブジェクトの型は生成後に変わりません。
      \end{block}
      \begin{block}{Goとの比較}
        Pythonは操作時の型検査や属性検索などを実行時に行います。メモリ量と速度差は、言語名だけでなく処理系、値、処理内容を固定して測ります。
      \end{block}
  \end{columns}
  \source{Python 3 Language Reference, “Data model”}
\end{frame}

\begin{frame}[t,fragile]{ミニ実習：配列、スライス、マップの性能を測ります}
  \begin{columns}[T,onlytextwidth]
    \column{0.58\textwidth}
      \begin{lstlisting}[language=Go]
var sink int

func BenchmarkSliceIndex(b *testing.B) {
    xs := make([]int, 1<<16)
    b.ResetTimer()
    for i := 0; i < b.N; i++ {
        sink = xs[i&(len(xs)-1)]
    }
}

func BenchmarkMapLookup(b *testing.B) {
    m := make(map[int]int, 1<<16)
    for i := 0; i < 1<<16; i++ { m[i] = i }
    b.ResetTimer()
    for i := 0; i < b.N; i++ {
        sink = m[i&((1<<16)-1)]
    }
}
      \end{lstlisting}
    \column{0.38\textwidth}
      \begin{block}{実行と比較条件}
        \code{go test -bench=. -benchmem}を5回実行し、\code{ns/op}と\code{allocs/op}を記録します。
        要素数、キー分布、事前確保、CPU、Go版を揃えます。
      \end{block}
      \begin{block}{追加課題}
        \code{append}の初期容量を0と要素数に変え、割り当て回数を比べます。測定値は計算量の証明ではありません。
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}[t]{第3章の要点は、抽象化の裏側にある共有とコストです}
  \begin{columns}[T,onlytextwidth]
    \column{0.31\textwidth}
      \begin{block}{実行}

      \end{block}
    \column{0.31\textwidth}

\begin{frame}[t]{確認問題では、仕様・実装・測定を分けて説明します}
  \begin{enumerate}

    \item スタックとヒープを、値の寿命と管理方法から比較してください。

    \item ハッシュマップの平均$O(1)$には、どのような仮定が必要ですか。
    \item GoとPythonの速度を公平に比べるため、固定すべき条件を挙げてください。
  \end{enumerate}
\end{frame}
